### 2.4 Server Actions 구조

```typescript
// actions/api-actions.ts
'use server';

import { z } from 'zod';
import { ApiCreateSchema } from '@/types/api';

export async function createApi(formData: FormData) {
  const parsed = ApiCreateSchema.safeParse(Object.fromEntries(formData));
  if (!parsed.success) return { success: false, errors: parsed.error.flatten() };

  const apiKey = process.env.INTERNAL_API_KEY;     // 클라에 미노출
  const baseUrl = process.env.BACKEND_BASE_URL;    // 사내 K8s ClusterIP

  const res = await fetch(`${baseUrl}/admin/apis`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-API-Key': apiKey!,
      'Authorization': `Bearer ${await getAccessToken()}`,
    },
    body: JSON.stringify(parsed.data),
    cache: 'no-store',
  });
  return res.json();
}
```

#### 3.4.1 DataSourceManager (멀티 데이터소스 hot-swap)
```java
@Component
public class DataSourceManager {
    private final ConcurrentHashMap<String, HikariDataSource> pools = new ConcurrentHashMap<>();
    private final RoutingDataSource routingDataSource;

    public void register(String dsId, DataSourceProps props) {
        var newPool = createHikari(props);
        var oldPool = pools.put(dsId, newPool);
        Map<Object, Object> targets = new HashMap<>(pools);
        routingDataSource.setTargetDataSources(targets);
        routingDataSource.afterPropertiesSet();   // 라우팅 갱신
        if (oldPool != null) gracefulDrain(oldPool);
    }

    private void gracefulDrain(HikariDataSource pool) {
        // 1. 신규 connection 차단 (suspend)
        pool.getHikariPoolMXBean().suspendPool();
        // 2. in-flight 종료 대기 (max 30s)
        // 3. close
        pool.close();
    }
}
```

#### 3.4.3 GatewayController (외부 호출 catch-all)
```java
@RestController
@RequestMapping("/api/**")
public class GatewayController {
    @RequestMapping
    public ResponseEntity<?> handle(HttpServletRequest req,
                                    @RequestBody(required = false) Map<String, Object> body) {
        var path = extractPath(req.getRequestURI());          // "/api/sample-user-info" → "sample-user-info"
        var api = apiLookupService.findByPath(path);          // Redis 캐시
        if (api == null) return ResponseEntity.notFound().build();

        if (!api.getMethod().equals(req.getMethod()))
            return error(405, "METHOD_NOT_ALLOWED");

        var params = mergeParams(req, body);
        try {
            RoutingDataSource.setDsId(api.getDsId());
            var result = dynamicSqlExecutor.execute(api, params);
            var masked = maskingService.apply(result, api.getResponseColumns());
            return ResponseEntity.ok(Map.of("result", masked, "count", masked.size()));
        } finally {
            RoutingDataSource.clear();
            callHistoryProducer.send(buildHistory(req, api, ...));
        }
    }
}
```

#### 3.4.4 ApiKeyAuthFilter (4단 검증)
```java
@Component
public class ApiKeyAuthFilter extends OncePerRequestFilter {
    @Override
    protected void doFilterInternal(HttpServletRequest req, HttpServletResponse res, FilterChain chain)
            throws ServletException, IOException {
        if (!req.getRequestURI().startsWith("/api/")) { chain.doFilter(req, res); return; }

        var key = req.getHeader("certification-key");
        var ip  = ipUtil.extractClientIp(req);
        var path = extractPath(req.getRequestURI());

        var sys = certKeyService.lookup(key);  // Redis cached
        if (sys == null) { reject(res, 401, "AUTH_KEY_INVALID"); return; }
        if (!sys.allowsIp(ip))                { reject(res, 403, "IP_NOT_ALLOWED"); return; }
        if (!sys.isActiveAt(LocalDateTime.now())) { reject(res, 403, "KEY_EXPIRED"); return; }
        if (!sys.mapsApiPath(path))           { reject(res, 403, "API_NOT_MAPPED"); return; }

        req.setAttribute("ext-sys-id", sys.getId());
        chain.doFilter(req, res);
    }
}
```

#### 3.4.5 DynamicSqlExecutor (MyBatis 동적 실행)
- MyBatis `SqlSessionFactory` 의 `Configuration.addMappedStatement` 로 런타임 매퍼 등록 또는
- `JdbcTemplate.queryForList(sql, params)` + named-parameter 로 단순 실행 (권장)
- 파라미터 `#{param}` → `:param` 변환 후 NamedJdbcTemplate 사용

```java
@Service
public class DynamicSqlExecutor {
    private final NamedParameterJdbcTemplate jdbcTemplate;

    public List<Map<String, Object>> execute(ApiDefinition api, Map<String, Object> params) {
        var sql = SqlValidator.validate(api.getSqlSource());      // 화이트리스트
        return jdbcTemplate.queryForList(sql, params);
    }
}
```

#### 3.4.7 SQL 검증 (등록 시점)

> 정책: 09 §4.2 와 일치. 첫 토큰 6종(SELECT/INSERT/UPDATE/DELETE/MERGE/CALL) 만 허용, **`BEGIN`/`DECLARE`/`EXECUTE IMMEDIATE` 는 본문에서도 차단**, CALL 은 등록된 procedure 화이트리스트 적용.

```java
@Service
public class SqlValidator {
    private static final Pattern WHITELIST = Pattern.compile(
        "^\\s*(SELECT|INSERT|UPDATE|DELETE|MERGE|CALL)\\b", Pattern.CASE_INSENSITIVE);

    private static final Set<String> BLOCKED = Set.of(
        "DROP", "TRUNCATE", "ALTER", "CREATE", "GRANT", "REVOKE",
        "BEGIN", "DECLARE", "EXECUTE IMMEDIATE",
        "DBMS_", "UTL_", "SYS.", "DBA_");

    private static final Pattern CALL_PROC = Pattern.compile(
        "^\\s*CALL\\s+([A-Z_][A-Z0-9_$.]*)\\s*\\(", Pattern.CASE_INSENSITIVE);

    public String validate(String sql, ApiDefinition api) {
        if (!WHITELIST.matcher(sql).find())
            throw new SqlNotAllowedException("FIRST_TOKEN_NOT_ALLOWED");
        var upper = sql.toUpperCase();
        for (var token : BLOCKED) {
            if (upper.contains(token))
                throw new SqlNotAllowedException("BLOCKED_TOKEN: " + token);
        }
        var m = CALL_PROC.matcher(sql);
        if (m.find() && !api.isCallProcedureAllowed(m.group(1)))
            throw new SqlNotAllowedException("CALL_PROC_NOT_REGISTERED: " + m.group(1));
        return sql;
    }

    /** DML 은 EXPLAIN PLAN, CALL/PLSQL 은 DBMS_SQL.PARSE 로 분기. */
    public void verifySyntax(String sql, String dsId) {
        if (CALL_PROC.matcher(sql).find())
            dbmsSqlParse(sql, dsId);
        else
            explainPlan(sql, dsId);
    }
}
```
